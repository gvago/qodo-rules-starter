Exercise 1 · Evidence

Find one to three real findings

The compiler accepts this code and the happy-path run prints success. What should review still catch?

#include "sensor_buffer.h"
#include <stdlib.h>

/* Ingest a burst of raw readings coming off the CAN receive path. */
int sensor_ingest(sensor_buffer_t *buf, const uint16_t *raw, size_t n) {
    uint16_t *scaled = malloc(n * sizeof(uint16_t));

    for (size_t i = 0; i < n; i++) {
        scaled[i] = (uint16_t)(raw[i] * 4u);
        sensor_buffer_push(buf, scaled[i]);
    }

    free(scaled);
    return SENSOR_OK;
}
Context
#define SENSOR_OK         0
#define SENSOR_ERR_RANGE (-1)
#define SENSOR_ERR_NULL  (-2)

int sensor_buffer_push(sensor_buffer_t *buf, uint16_t sample);

Assume sensor_buffer_push can return SENSOR_ERR_NULL or SENSOR_ERR_RANGE.
